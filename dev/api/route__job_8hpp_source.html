193struct RouteJobResult {
194 route::RouteResult result;
195 nlohmann::json items; // new copper in the --emit-items layout (layer and net by name, nm)
196 nlohmann::json summary; // the --json layout
197 std::string viewer_url; // empty when not viewing
198 int unconnected_after_refill = -1; // with soft zones: what KiCad's refill would leave unconnected
199 // 0 when everything planned was routed (and, with soft zones, nothing is unconnected after the refill).
200 int exit_code() const { return unconnected_after_refill >= 0 ? (unconnected_after_refill == 0 ? 0 : 3) : result.routed == result.connections ? 0 : 3; }
201 // Why exit_code() is what it is (issue 43): exit 3 conflates connections the router left
202 // open with pours KiCad's zone refill would leave unconnected on an otherwise complete
203 // board. The route JSON summary carries this discriminator so callers can tell them apart.
204 // The discriminator is the routed count, not the refill count (issue 54):
205 // unconnected_after_refill counts every unconnected item on the refilled board, the pads
206 // of connections the router never routed included, so keying the reason on it labels any
207 // soft-zones job with genuine opens "unconnected_after_refill". Only a job that routed
208 // everything it planned and still exits 3 earns that reason.
209 const char* exit_reason() const {
210 if (exit_code() == 0) return "complete";
211 return result.routed < result.connections ? "open_connections" : "unconnected_after_refill";
212 }
213};

219// The layer-count advisor (`route --try-layers`, doc 05 §53, M20 C3): routes the board with each copper layer count in `counts`,
220// fewest first, and stops at the first that completes (exit code 0: everything routed and, with soft zones, nothing unconnected after
221// the refill). Its board is written to `job.out` and its summary returned, with a "layers" object listing every count tried. A count
222// the board cannot take (inner layers that are not empty) is reported and skipped. When none completes, the result of the count that
223// left the fewest connections open is returned (the first such count on a tie).
224RouteJobResult run_layer_advisor(RouteJob job, std::vector<int> counts);
225
226// `route --finish` (doc 20 §2.12, D196): rungs that each start from the board the rung before wrote, so what is routed stays routed and
227// only the open connections are searched. Rung 1 is the job as given. Rung 2 routes what is open with the rules fitted to the board's own
228// copper (`fit_rules`, or `fit_rules_from` when the job names a reference) and the repair windows along each connection's bare-board route
229// and round its ends. Each entry of `extra_layers` (e.g. 2, 4) is one more rung with that many copper layers added to the input's count.
230// With `fresh_rung`, a rung after the second routes the board as given with the fitted rules from the start (not on top) and is kept when it
231// leaves fewer open. With `masks_rung`, a rung before the layers lets tracks through pad solder-mask openings (RouterOptions::soft_pad_masks), and the layer
232// rungs keep that. The ladder stops at the first rung with nothing open; a rung that does not lower the open count is discarded (rule 4). The summary's
233// "finish" lists every rung with what it relaxed. Needs an output path. With `guard_walls` (DR-22), a routing rung is kept only when
234// open + walled-in pins improves over the best board so far: a rung's own copper can wall the pins it leaves open in, and no later rung
235// can reach a walled pin, so the plain open-count rule keeps boards that are worse than they look.
236RouteJobResult run_finish(RouteJob job, std::vector<int> extra_layers, bool masks_rung = false, bool fresh_rung = false, bool chains_rung = false,
237 bool guard_walls = false, bool rip_walls = false);

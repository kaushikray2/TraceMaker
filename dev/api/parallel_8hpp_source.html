19#pragma once
20// The process-wide worker budget behind --threads as a hard global cap (Stage 2B #19).
21//
22// The app sets the cap once per process (a route job sets it from its thread count); 0 means no
23// cap was set and every stage keeps its historical behaviour. The portfolio's variant workers
24// are the primary workers: route_portfolio registers how many are alive, and secondary stages
25// (zone refill's per-layer tasks) may only add threads with what the cap leaves free, so the
26// process never runs more concurrent workers than the cap. The main thread and the live-log
27// writer are not workers; the documented OS-level thread peak is cap + 2.
28#include <algorithm>
29#include <atomic>
30#include <climits>

47inline void set_cap(int n) { cap_slot().store(std::max(0, n)); }

49// RAII save/restore of the cap for one job's scope: the previous value (0 = no cap)
50// is restored on exit, so a process that runs more than one job -- the test binary,
51// batch mode, nested wrapper jobs -- never leaks one job's cap into the next.
52struct CapScope {
53 explicit CapScope(int n) : prev_(cap_slot().load()) { set_cap(n); }
54 ~CapScope() { set_cap(prev_); }
55 CapScope(const CapScope&) = delete;
56 CapScope& operator=(const CapScope&) = delete;
57
58 private:
59 int prev_;
60};
61inline int cap() { return cap_slot().load(); }
62inline int primary_active() { return primary_slot().load(); }
63
64// How many secondary worker threads a stage may add right now: unbounded when no cap is set,
65// otherwise what the cap leaves after the live primary workers (never negative).
66inline int secondary_allowance() {
67 const int c = cap();
68 if (c <= 0) return INT_MAX;
69 return std::max(0, c - primary_active());
70}
71
72// RAII claim of secondary worker slots under the cap. secondary_allowance() alone is a
73// read, not a reservation: concurrent secondary stages (zone refill runs inside every
74// portfolio variant's setup when the board has unfilled zones) would each spend the
75// same headroom and oversubscribe the cap. A claim atomically takes as many of `want`
76// as the cap leaves free after the live primary workers and other live claims, and
77// releases them on scope exit. With no cap set, count() is -1: unbounded, the
78// historical behaviour.
79struct SecondaryScope {
80 explicit SecondaryScope(int want) : n_(0) {
81 if (cap() <= 0) {
82 n_ = -1;
83 return;
84 }
85 int cur = secondary_slot().load();
86 for (;;) {
87 const int avail = cap() - primary_active() - cur;
88 const int claim = std::min(want, std::max(0, avail));
89 if (claim <= 0) return;
90 if (secondary_slot().compare_exchange_weak(cur, cur + claim)) {
91 n_ = claim;
92 return;
93 }
94 }
95 }
96 ~SecondaryScope() {
97 if (n_ > 0) secondary_slot().fetch_sub(n_);
98 }
99 SecondaryScope(const SecondaryScope&) = delete;
100 SecondaryScope& operator=(const SecondaryScope&) = delete;
101 int count() const { return n_; }
102
103 private:
104 int n_;
105};

107// RAII registration of a primary pool's workers, for the scope of a parallel run.
108struct PrimaryScope {
109 explicit PrimaryScope(int n) : n_(n > 0 ? n : 0) {

112 ~PrimaryScope() {

114 }

120};

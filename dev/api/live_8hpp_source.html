9#pragma once
10// Live status of this process for the progress site's live page (devsite /live, D165): what the process is doing, written as
11// JSON to <dir>/<pid>.json about once a second by a background thread (written to a temporary file and renamed, so a reader
12// never sees half a file). <dir> is $TM_LIVE_DIR, else $XDG_CACHE_HOME/tracemaker/live, else ~/.cache/tracemaker/live.
13// On by default; TM_LIVE=0 turns it off. A process that ends within its first second writes nothing.
14//
15// Write-only: the engine never reads any of it back, so it cannot change a result (rule 2). The router updates its slot with
16// relaxed atomic stores, which cost next to nothing next to a search; everything else takes a mutex a few times a second.
17#include <atomic>
18#include <string>
19
20namespace tmk::live {
21
22// One router's progress (a portfolio variant, by name), updated lock-free from the thread running it.
23struct Slot {
24 std::atomic<int> routed{0}, total{0};
25 std::atomic<long> work{0}, budget{0}; // search expansions of the run in progress and its work budget (0: none)
26 std::atomic<long> work_done{0}; // search expansions of the slot's finished runs
27 std::atomic<int> phase{0}; // 0 waiting, 1 setup, 2 first pass, 3 negotiation, 4 clean-up, 5 done
28 std::atomic<int> runs{0}; // router runs started in this slot (one per round)
29};
30
31// From main(): starts the writer thread, unless TM_LIVE=0. argv is recorded as the command line.
32void start(int argc, char** argv);
33// From main(): writes the final state with the exit code and stops the writer. Without it (std::exit) the last state says "ended".
34void finish(int exit_code);
35// True after start() when the status is being written.
36bool enabled();
37// The slot named `name`, created on first use; the pointer stays valid for the life of the process. nullptr when not enabled.
38Slot* slot(const std::string& name);
39// Sets one line of the status ("stage", "board", "round", ...), shown in the order first set. An empty value removes it.
40void field(const std::string& key, const std::string& value);
41// A point-in-time read of the status for in-process consumers (the route job's --heartbeat file): the stage
42// line and the furthest-along router slot. routed/total are -1 until a router reports them; all defaults
43// when live status is off.
44struct Progress {
45 std::string stage;
46 int routed = -1, total = -1;
47 long work = 0, budget = 0;
48};
49Progress progress();

19#pragma once
20// Flat open-addressing containers (Stage 2B: flat hash maps): a uint64 -> uint16 map and
21// an int64-pair set, both linear-probing with backward-shift-free growth rehashing.
22//
23// Linear probing over a power-of-two slot array with a splitmix64 finalizer for the probe
24// start, backward-shift deletion (no tombstones), growth at 70 % load. Built for the route
25// history map: ~1e8 finds per large job against std::unordered_map cost several seconds of
26// pointer chasing; a find here is one or two cache lines. Key 0 is a valid key (occupancy
27// is a separate flag, not a sentinel). Iteration is over raw slots for the one caller that
28// transforms values in place (the history decay sweep); slot order is unspecified and no
29// caller may depend on it.
30#include <cstddef>
31#include <cstdint>
32#include <vector>
33

36class FlatU64U16 {
37 public:
38 bool empty() const { return n_ == 0; }
39 std::size_t size() const { return n_; }

41 const std::uint16_t* find(std::uint64_t key) const {
42 if (slots_.empty()) return nullptr;
43 std::size_t i = probe(key);
44 return slots_[i].used ? &slots_[i].value : nullptr;
45 }
46 std::uint16_t& operator[](std::uint64_t key) {
47 if (slots_.empty() || (n_ + 1) * 10 > slots_.size() * 7) grow();
48 std::size_t i = probe(key);
49 if (!slots_[i].used) {
50 slots_[i].used = true;
51 slots_[i].key = key;
52 slots_[i].value = 0;
53 ++n_;
54 }
55 return slots_[i].value;
56 }

58 // Slot access for read-only sweeps (the heatmap build). Do NOT erase in place during a
59 // hand-rolled sweep: backward shift moves an entry from a higher slot into the erased
60 // one, and across the ring wrap (last slot -> slot 0) that entry has already been
61 // visited, so it would be processed twice. Use decay_half() for in-place decay.
62 std::size_t slot_count() const { return slots_.size(); }
63 bool slot_used(std::size_t i) const { return slots_[i].used; }
64 std::uint16_t& value_at(std::size_t i) { return slots_[i].value; }
65 std::uint64_t key_at(std::size_t i) const { return slots_[i].key; }
66 void erase_at(std::size_t i) {
67 const std::size_t mask = slots_.size() - 1;
68 std::size_t hole = i;
69 slots_[hole].used = false;
70 --n_;
71 std::size_t j = hole;
72 for (;;) {
73 j = (j + 1) & mask;
74 if (!slots_[j].used) return;
75 const std::size_t home = mix(slots_[j].key) & mask;
76 // Move slot j into the hole when its home is not in the cyclic interval (hole, j].
77 const bool home_after_hole = hole < j ? (home > hole && home <= j) : (home > hole || home <= j);
78 if (!home_after_hole) {
79 slots_[hole] = slots_[j];
80 slots_[j].used = false;
81 hole = j;
82 }
83 }
84 }
85 void clear() {
86 for (auto& s : slots_) s.used = false;
87 n_ = 0;
88 }

90 // Halve every value in place, erasing entries whose value reaches 0 -- the history
91 // decay. Exact-once: the sweep starts just after an empty slot (one always exists at
92 // the <= 70 % load the growth rule keeps) and walks the ring from there. Backward-shift
93 // deletion never moves an entry across an empty slot, so every entry an erase shifts
94 // into the sweep's path comes from its own cluster, ahead of the sweep -- never from
95 // behind it, which is exactly what goes wrong across the wrap in a sweep from slot 0.
96 void decay_half() {
97 if (slots_.empty()) return;
98 const std::size_t n = slots_.size();
99 std::size_t s0 = 0;
100 while (s0 < n && slots_[s0].used) ++s0; // the empty slot anchoring the sweep
101 if (s0 == n) return; // a full table cannot occur under the growth rule
102 for (std::size_t k = 1; k < n; ++k) {
103 std::size_t idx = s0 + k;
104 if (idx >= n) idx -= n;
105 if (!slots_[idx].used) continue;
106 const std::uint16_t hv = static_cast<std::uint16_t>(slots_[idx].value / 2);
107 if (hv == 0) {
108 erase_at(idx);
109 --k; // re-examine idx: the entry shifted into it is unvisited
110 } else {
111 slots_[idx].value = hv;
112 }
113 }
114 }

116 private:
117 struct Slot {
118 std::uint64_t key = 0;
119 std::uint16_t value = 0;
120 bool used = false;
121 };
122 static std::uint64_t mix(std::uint64_t k) {
123 k ^= k >> 30;
124 k *= 0xbf58476d1ce4e5b9ULL;
125 k ^= k >> 27;
126 k *= 0x94d049bb133111ebULL;
127 k ^= k >> 31;
128 return k;
129 }
130 std::size_t probe(std::uint64_t key) const {
131 const std::size_t mask = slots_.size() - 1;
132 std::size_t i = mix(key) & mask;
133 while (slots_[i].used && slots_[i].key != key) i = (i + 1) & mask;
134 return i;
135 }
136 void grow() {
137 std::vector<Slot> old = std::move(slots_);
138 slots_.assign(old.empty() ? 1024 : old.size() * 2, Slot{});
139 const std::size_t mask = slots_.size() - 1;
140 for (const auto& s : old) {
141 if (!s.used) continue;
142 std::size_t i = mix(s.key) & mask;
143 while (slots_[i].used) i = (i + 1) & mask;
144 slots_[i] = s;
145 }
146 }
147 std::vector<Slot> slots_;
148 std::size_t n_ = 0;
149};

151// Flat open-addressing set of int64 pairs (the learned_block shape): same probing core as
152// FlatU64U16, membership and insert only (that caller never erases or iterates). The probe
153// start mixes both halves, matching the splitmix pair hash the unordered_set used.
154class FlatPairSet {
155 public:
156 bool empty() const { return n_ == 0; }
157 std::size_t size() const { return n_; }
158 bool contains(std::int64_t a, std::int64_t b) const {
159 if (slots_.empty()) return false;
160 const std::size_t mask = slots_.size() - 1;
161 std::size_t i = mix(static_cast<std::uint64_t>(a) ^ mix(static_cast<std::uint64_t>(b))) & mask;
162 while (slots_[i].used && (slots_[i].a != a || slots_[i].b != b)) i = (i + 1) & mask;
163 return slots_[i].used;
164 }
165 void insert(std::int64_t a, std::int64_t b) {
166 if (slots_.empty() || (n_ + 1) * 10 > slots_.size() * 7) grow();
167 const std::size_t mask = slots_.size() - 1;
168 std::size_t i = mix(static_cast<std::uint64_t>(a) ^ mix(static_cast<std::uint64_t>(b))) & mask;
169 while (slots_[i].used && (slots_[i].a != a || slots_[i].b != b)) i = (i + 1) & mask;
170 if (!slots_[i].used) {
171 slots_[i].used = true;
172 slots_[i].a = a;
173 slots_[i].b = b;
174 ++n_;
175 }
176 }

180 std::int64_t a = 0;
181 std::int64_t b = 0;
